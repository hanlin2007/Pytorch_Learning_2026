import torch.optim as optim
from torch.optim.lr_scheduler import CosineAnnealingLR, ReduceLROnPlateau
from torch.utils.data import DataLoader
import torchvision

# -------------------- 超参数配置类 --------------------
class Config:
    # 数据参数
    data_path = './data'
    batch_size = 128                # 增大batch size以配合BN
    num_workers = 4 if os.name != 'nt' else 0  # Windows建议0

    # 数据增强参数

    momentum = 0.9                     # SGD动量
    optimizer_type = 'adamw'           # 'adamw', 'sgd'
    scheduler_type = 'cosine'          # 'cosine', 'plateau', 'step'
    warmup_epochs = 5                  # 学习率预热epoch数
    early_stop_patience = 10           # 早停耐心值
    grad_clip_norm = 1.0                # 梯度裁剪范数

    # 设备
    device = 'cuda' if torch.cuda.is_available() else 'cpu'

    # 保存路径

config = Config()

if config.device == 'cuda':

# -------------------- 数据准备（含增强和归一化） --------------------
def prepare_data():
    # 基础变换：转Tensor
    base_transform = [transforms.ToTensor()]

    # 训练集数据增强
    train_transforms = []
    if config.use_augmentation:
        train_transforms.extend([
            transforms.RandomAffine(
                degrees=config.rotation_deg,
                translate=(config.translate, config.translate),
                scale=config.scale
            ),
        ])
    # 标准化 (MNIST官方推荐均值0.1307，标准差0.3081)
    normalize = transforms.Normalize((0.1307,), (0.3081,))
    train_transforms.extend([transforms.ToTensor(), normalize])
    test_transforms = [transforms.ToTensor(), normalize]

    train_transform = transforms.Compose(train_transforms)
    test_transform = transforms.Compose(test_transforms)

    print("\n加载MNIST数据集...")
    start_time = time.time()

    train_dataset = torchvision.datasets.MNIST(
        root=config.data_path, train=True, download=True, transform=train_transform
    )
    test_dataset = torchvision.datasets.MNIST(
        root=config.data_path, train=False, download=True, transform=test_transform
    )

    load_time = time.time() - start_time
    print(f"加载完成! 耗时: {load_time:.2f}秒")
    print(f"训练集: {len(train_dataset)} 样本, 测试集: {len(test_dataset)} 样本")

    train_loader = DataLoader(
        train_dataset,
        batch_size=config.batch_size,
        shuffle=True,
        num_workers=config.num_workers,
        pin_memory=config.pin_memory
    )
    test_loader = DataLoader(
        test_dataset,
        batch_size=config.batch_size,
        shuffle=False,
        num_workers=config.num_workers,
        pin_memory=config.pin_memory

    return train_loader, test_loader

# -------------------- 模型定义（含残差网络示例） --------------------
# 权重初始化函数
def init_weights(m):
    if isinstance(m, nn.Linear):
        nn.init.kaiming_normal_(m.weight, mode='fan_out', nonlinearity='relu')
        if m.bias is not None:
            nn.init.constant_(m.bias, 0)
    elif isinstance(m, nn.Conv2d):
        nn.init.kaiming_normal_(m.weight, mode='fan_out', nonlinearity='relu')
        if m.bias is not None:
            nn.init.constant_(m.bias, 0)
    elif isinstance(m, nn.BatchNorm2d) or isinstance(m, nn.BatchNorm1d):
        nn.init.constant_(m.weight, 1)
        nn.init.constant_(m.bias, 0)

# 残差块定义（用于小型ResNet）
class ResidualBlock(nn.Module):
    def __init__(self, in_channels, out_channels, stride=1):
        super().__init__()
        self.conv1 = nn.Conv2d(in_channels, out_channels, kernel_size=3, stride=stride, padding=1, bias=False)
        self.bn1 = nn.BatchNorm2d(out_channels)
        self.relu = nn.ReLU(inplace=True)
        self.conv2 = nn.Conv2d(out_channels, out_channels, kernel_size=3, stride=1, padding=1, bias=False)
        self.bn2 = nn.BatchNorm2d(out_channels)

        self.shortcut = nn.Sequential()
        if stride != 1 or in_channels != out_channels:
            self.shortcut = nn.Sequential(
                nn.Conv2d(in_channels, out_channels, kernel_size=1, stride=stride, bias=False),
                nn.BatchNorm2d(out_channels)
            )

    def forward(self, x):
        out = self.relu(self.bn1(self.conv1(x)))
        out = self.bn2(self.conv2(out))
        out += self.shortcut(x)
        out = self.relu(out)
        return out

# 小型ResNet（适用于MNIST）
class SmallResNet(nn.Module):
    def __init__(self, num_classes=10):
        super().__init__()
        self.in_channels = 32
        self.conv1 = nn.Conv2d(1, 32, kernel_size=3, stride=1, padding=1, bias=False)
        self.bn1 = nn.BatchNorm2d(32)
        self.relu = nn.ReLU(inplace=True)

        self.layer1 = self._make_layer(32, 2, stride=1)
        self.layer2 = self._make_layer(64, 2, stride=2)
        self.layer3 = self._make_layer(128, 2, stride=2)

        self.avg_pool = nn.AdaptiveAvgPool2d((1, 1))
        self.fc = nn.Linear(128, num_classes)
        self.dropout = nn.Dropout(config.dropout_rate)

        # 应用权重初始化
        self.apply(init_weights)

    def _make_layer(self, out_channels, num_blocks, stride):
        strides = [stride] + [1]*(num_blocks-1)
        layers = []
        for s in strides:
            layers.append(ResidualBlock(self.in_channels, out_channels, s))
            self.in_channels = out_channels
        return nn.Sequential(*layers)

    def forward(self, x):
        out = self.relu(self.bn1(self.conv1(x)))
        out = self.layer1(out)
        out = self.layer2(out)
        out = self.layer3(out)
        out = self.avg_pool(out)
        out = out.view(out.size(0), -1)
        out = self.dropout(out)
        out = self.fc(out)
        return out

# 备选简单CNN（含BatchNorm）
class SimpleCNN(nn.Module):
    def __init__(self):
        super().__init__()
        self.features = nn.Sequential(
            nn.Conv2d(1, 32, kernel_size=3, padding=1),
            nn.BatchNorm2d(32),
            nn.ReLU(inplace=True),
            nn.MaxPool2d(2),
            nn.Conv2d(32, 64, kernel_size=3, padding=1),
            nn.BatchNorm2d(64),
            nn.ReLU(inplace=True),
            nn.MaxPool2d(2),
        )
        self.classifier = nn.Sequential(
            nn.Dropout(config.dropout_rate),
            nn.Linear(64 * 7 * 7, 256),
            nn.BatchNorm1d(256),
            nn.ReLU(inplace=True),
            nn.Dropout(config.dropout_rate),
            nn.Linear(256, config.num_classes)
        )
        self.apply(init_weights)

    def forward(self, x):
        x = self.features(x)
        x = x.view(x.size(0), -1)
        x = self.classifier(x)
        return x

def create_model():
    if config.model_type == 'resnet':
        model = SmallResNet(num_classes=config.num_classes)
    elif config.model_type == 'cnn':
        model = SimpleCNN()
    else:  # MLP（保留原始MLP但增加BN）
        model = nn.Sequential(
            nn.Flatten(),
            nn.Linear(784, config.hidden_size),
            nn.BatchNorm1d(config.hidden_size),
            nn.ReLU(),
            nn.Dropout(config.dropout_rate),
            nn.Linear(config.hidden_size, config.hidden_size),
            nn.BatchNorm1d(config.hidden_size),
            nn.ReLU(),
            nn.Dropout(config.dropout_rate),
            nn.Linear(config.hidden_size, config.num_classes)
        )
        model.apply(init_weights)
    return model.to(config.device)

# -------------------- 训练工具函数 --------------------
def train_one_epoch(model, loader, criterion, optimizer, scheduler, epoch, scaler=None):
    model.train()
    running_loss = 0.0
    correct = 0
    total = 0

    pbar = tqdm(loader, desc=f"Epoch {epoch+1} Train", leave=False)
    for images, labels in pbar:
        images, labels = images.to(config.device), labels.to(config.device)

        # 混合精度训练（可选）
        if scaler:
            with torch.cuda.amp.autocast():
                outputs = model(images)
                loss = criterion(outputs, labels)
            scaler.scale(loss).backward()
            scaler.unscale_(optimizer)
            nn.utils.clip_grad_norm_(model.parameters(), config.grad_clip_norm)  # 梯度裁剪
            scaler.step(optimizer)
            scaler.update()
        else:
            outputs = model(images)
            loss = criterion(outputs, labels)
            optimizer.zero_grad()
            loss.backward()
            nn.utils.clip_grad_norm_(model.parameters(), config.grad_clip_norm)
            optimizer.step()

        optimizer.zero_grad()  # 注意：如果已经在上面的分支中调用过step，此处可省略；这里为清晰保留

        running_loss += loss.item()
        _, predicted = outputs.max(1)
        total += labels.size(0)
        correct += predicted.eq(labels).sum().item()

        pbar.set_postfix({'loss': loss.item(), 'acc': 100.*correct/total})

    avg_loss = running_loss / len(loader)
    acc = 100. * correct / total
    return avg_loss, acc

def evaluate(model, loader, criterion):
    model.eval()
    running_loss = 0.0
    correct = 0
    total = 0
    with torch.no_grad():
        for images, labels in loader:
            images, labels = images.to(config.device), labels.to(config.device)
            outputs = model(images)
            loss = criterion(outputs, labels)
            running_loss += loss.item()
            _, predicted = outputs.max(1)
            total += labels.size(0)
            correct += predicted.eq(labels).sum().item()
    avg_loss = running_loss / len(loader)
    acc = 100. * correct / total
    return avg_loss, acc

# -------------------- 学习率预热调度器包装 --------------------
def get_optimizer_and_scheduler(model):
    # 优化器
    if config.optimizer_type == 'adamw':
        optimizer = optim.AdamW(model.parameters(), lr=config.learning_rate,
                                weight_decay=config.weight_decay)
    else:  # sgd
        optimizer = optim.SGD(model.parameters(), lr=config.learning_rate,
                              momentum=config.momentum, weight_decay=config.weight_decay)

    # 调度器
    if config.scheduler_type == 'cosine':
        scheduler = CosineAnnealingLR(optimizer, T_max=config.epochs - config.warmup_epochs)
    elif config.scheduler_type == 'plateau':
        scheduler = ReduceLROnPlateau(optimizer, mode='min', factor=0.5, patience=5)
    else:
        scheduler = optim.lr_scheduler.StepLR(optimizer, step_size=10, gamma=0.5)

    # 学习率预热（通过自定义LambdaLR实现）
    if config.warmup_epochs > 0:
        def warmup_lambda(epoch):
            if epoch < config.warmup_epochs:
                return (epoch + 1) / config.warmup_epochs
            return 1.0
        warmup_scheduler = optim.lr_scheduler.LambdaLR(optimizer, lr_lambda=warmup_lambda)
        # 将预热与主调度器结合（可使用SequentialLR）
        schedulers = [warmup_scheduler, scheduler]
        milestones = [config.warmup_epochs]
        scheduler = optim.lr_scheduler.SequentialLR(optimizer, schedulers, milestones)

    return optimizer, scheduler

# -------------------- 主训练流程 --------------------
def main():
    print("="*50)
    print("超参数调优与训练技巧综合示例 (MNIST)")
    print("="*50)

    # 1. 准备数据
    train_loader, test_loader = prepare_data()

    # 2. 创建模型
    model = create_model()
    print(f"\n模型结构:\n{model}")
    total_params = sum(p.numel() for p in model.parameters())
    print(f"模型参数量: {total_params:,}")

    # 3. 损失函数
    criterion = nn.CrossEntropyLoss()

    # 4. 优化器和调度器
    optimizer, scheduler = get_optimizer_and_scheduler(model)

    # 5. 混合精度（可选，仅当CUDA可用）
    scaler = torch.cuda.amp.GradScaler() if config.device == 'cuda' else None

    # 6. 记录与早停
    best_acc = 0.0
    epochs_no_improve = 0
    train_losses, test_losses = [], []
    train_accs, test_accs = [], []

    print("\n开始训练...")
    total_start = time.time()

    for epoch in range(config.epochs):
        # 训练
        train_loss, train_acc = train_one_epoch(model, train_loader, criterion,
                                                 optimizer, scheduler, epoch, scaler)
        # 测试
        test_loss, test_acc = evaluate(model, test_loader, criterion)

        # 记录
        train_losses.append(train_loss)
        test_losses.append(test_loss)
        train_accs.append(train_acc)
        test_accs.append(test_acc)

        # 更新调度器（根据调度器类型）
        if isinstance(scheduler, optim.lr_scheduler.ReduceLROnPlateau):
            scheduler.step(test_loss)
        else:
            scheduler.step()

        # 打印当前学习率
        current_lr = optimizer.param_groups[0]['lr']

        print(f"\nEpoch {epoch+1}/{config.epochs} | LR: {current_lr:.2e} | "
              f"Train Loss: {train_loss:.4f} Train Acc: {train_acc:.2f}% | "
              f"Test Loss: {test_loss:.4f} Test Acc: {test_acc:.2f}%")

        # 保存最佳模型
        if test_acc > best_acc:
            best_acc = test_acc
            torch.save(model.state_dict(), config.model_save_path)
            print(f"✓ 新最佳模型保存 (准确率: {best_acc:.2f}%)")
            epochs_no_improve = 0
        else:
            epochs_no_improve += 1

        # 早停
        if epochs_no_improve >= config.early_stop_patience:
            print(f"早停触发：连续 {config.early_stop_patience} 轮测试准确率未提升")
            break

    total_time = time.time() - total_start
    print(f"\n训练完成！总耗时: {total_time:.2f}秒 ({total_time/60:.2f}分钟)")
    print(f"最佳测试准确率: {best_acc:.2f}%")

    # 加载最佳模型并最终评估
    model.load_state_dict(torch.load(config.model_save_path))
    final_loss, final_acc = evaluate(model, test_loader, criterion)
    print(f"最终测试准确率（最佳模型）: {final_acc:.2f}%")

if __name__ == '__main__':
    main()

# 图像识别进阶拓展

## （一）两位数字识别→多位数字识别
### 方案 B：拆成“十位 + 个位”两个任务（降低难度）

适合：

- 数据量一般般；
    
- 想提高一点准确率、训练更稳定。
    

思路是把一个两位数识别，拆成两步：

1. **第一步：识别“十位数字”（0–9）**​
    
2. **第二步：识别“个位数字”（0–9）**
    

模型上可以有两种实现方式：

- **多任务学习（共享 backbone）**​
    
    - 一个特征提取网络（CNN 等）提特征；
        
    - 后面分两支：
        
        - 分支1：输出 10 类，预测十位；
            
        - 分支2：输出 10 类，预测个位；
            
        
    - 总损失 = loss(十位) + loss(个位)。
        
    
- **两阶段流水线**​
    
    - 先用一个模型判断“十位”；
        
    - 再用一个模型判断“个位”；
        
    - 好处是逻辑清晰，坏处是两个模型、部署稍麻烦。
        
    

这样做的好处：

- 每一部分都是熟悉的 10 分类问题，难度大幅降低；
    
- 即使某些两位数样本很少，只要十位、个位各自的样本够，也能学得好；
    
- 模型更容易收敛，调试也更方便。
    

---

### 方案 C：字符级识别（当成两个字符序列）

如果你愿意再往前一步，可以把整个 0–99 识别看成：

**“识别一串长度 ≤ 2 的数字字符”**，也就是 OCR 里的“字符序列识别”。

- 输入一张图片；
    
- 模型输出一个长度为 2 的序列（不足的前面补“空白”或“前导零”）；
    
- 每个位置都是一个 10 分类（0–9）。
    

这种方式更适合以后扩展到“任意位数”的场景，比如 0–999、甚至带小数点的数字。

## （二）多任务图像识别

这是一个典型的**多任务学习（Multi-Task Learning, MTL）问题，包含两个子任务：

1. **图像分类**：判断背景是猫还是狗（2 类）。
    
2. **数字识别**：识别屏幕上的数字（0–9，10 类）。
    

这两个任务共享底层视觉特征（比如边缘、纹理、形状），但各自有独立的高层语义，非常适合用**共享主干 + 多任务头**的网络结构来解决。

### 1. 网络结构设计

核心思路：

- **Backbone（主干网络）**：提取通用图像特征。
    
- **Task-Specific Heads（任务头）**：在共享特征基础上，分别处理两个任务。
    

### 结构示意

```
输入图像 (H×W×C)
      ↓
[ Backbone 卷积网络 ]  ← 共享特征提取（如 ResNet、MobileNet、CNN）
      ↓
共享特征图 (H'×W'×D)
      ↓
┌───────────────┬───────────────┐
│  背景分类头   │  数字识别头   │
│  (猫/狗 2类)  │  (0–9 10类)  │
└───────────────┴───────────────┘
      ↓               ↓
  背景概率        数字概率
```

### 具体实现建议

- **Backbone**：
    
    - 轻量级：MobileNetV2/V3、ShuffleNet（移动端/实时场景）。
        
    - 精度优先：ResNet18/34、EfficientNet-B0（服务器/云端）。
        
    - 输入尺寸：比如 224×224×3，可根据实际图片大小调整。
        
    
- **背景分类头**：
    
    - 在共享特征图上做全局平均池化（GAP）→ 全连接层 → Softmax(2)。
        
    - 或直接用 1×1 卷积压缩通道后接 GAP + FC。
        
    
- **数字识别头**：
    
    - 如果是**单个数字居中**：类似背景头，GAP → FC → Softmax(10)。
        
    - 如果是**数字可能出现在任意位置**：可以用 CNN + RNN/Transformer 做序列识别（CRNN 结构），或者直接在全图特征上接一个小的分类头（假设数字总是清晰可见且位置相对固定）。
        
    
### 2. 训练技巧与策略

### （1）损失函数设计

两个任务的损失要加权求和：

Ltotal​=α⋅Lbg​+β⋅Ldigit​

- Lbg​：背景分类的交叉熵损失（2 类）。
    
- Ldigit​：数字识别的交叉熵损失（10 类）。
    
- α,β：平衡系数，初始可设为 1.0，根据任务难度调整。
    
    - 如果背景分类容易（准确率高但提升慢），可以降低 α；
        
    - 如果数字识别难，可以提高 β。
        
    

**注意**：两个任务的学习难度可能不同，需要监控各自验证集指标，避免一个任务主导训练。

### （2）数据准备与增强

- **数据组织**：
    
    - 每条数据包含：图像 + 背景标签（猫/狗）+ 数字标签（0–9）。
        
    - 背景图和数字图可以合成（比如把数字贴在猫/狗图片的某个位置），也可以收集真实场景数据。
        
    
- **数据增强**：
    
    - 通用增强：随机裁剪、缩放、旋转、亮度/对比度变化（模拟不同光照）。
        
    - 针对数字干扰：轻微模糊、噪声、运动模糊（模拟摄像头抖动）、遮挡（模拟猫爪、狗尾巴挡住数字）。
        
    - 针对背景：猫/狗图片本身的颜色抖动、风格迁移（但不要破坏数字可读性）。
        
    
### （3）训练策略

1. **预训练 + 微调**：
    
    - 先用大规模图像分类数据集（如 ImageNet）预训练 Backbone。
        
    - 冻结 Backbone 前几层，只训练后面的层和任务头，稳定初期训练。
        
    - 逐步解冻更多层，联合微调所有参数。
        
    
2. **分阶段训练**（可选）：
    
    - 第一阶段：只训练背景分类任务，让 Backbone 学会区分猫/狗。
        
    - 第二阶段：解锁数字识别任务，联合训练两个任务。
        
    - 好处：避免数字任务在早期干扰背景特征的提取。
        
    
3. **学习率调度**：
    
    - 使用余弦退火或 ReduceLROnPlateau，根据验证集总损失调整学习率。
        
    - 两个任务头部可以用稍大的学习率（因为它们是从头训练的），Backbone 用较小学习率（因为是预训练的）。
        
    
### （4）评估与调试

- **单独评估每个任务**：
    
    - 背景分类准确率（猫/狗）。
        
    - 数字识别准确率（0–9）。
        
    - 不要只看总损失，要看每个任务的独立指标。
        
    
- **可视化分析**：
    
    - 对错误样本进行分析：是背景误判导致数字错？还是数字本身难认？
        
    - 可以用 Grad-CAM 等方法查看 Backbone 关注的区域，确认模型是否同时关注了背景和数字。
        
    
### （5）进阶技巧（可选）

- **动态权重调整**：
    
    使用不确定性加权（Uncertainty Weighting）自动学习 α和 β，让模型自己决定每个任务的重要性。
    
- **任务注意力机制**：
    
    在共享特征后加入注意力模块，让网络自动增强对数字区域的响应（尤其当数字很小或被遮挡时）。
    
- **对抗训练**：
    
    加入一个小判别器，试图区分特征来自猫还是狗背景，强迫 Backbone 提取更“中性”的特征，有利于数字识别。
    

## （三）少样本数字识别（Few-shot MNIST）

**任务**：每个数字只有 1~5 张训练样本，测试集正常。

**目标**：模型能快速适应新数字类别。

**创新点**：数据稀缺条件下的泛化能力。

---

### 一、先救数据：让有限样本发挥最大作用

### 1. 数据增强要“狠”一点

小数据最怕过拟合，增强就是“人造数据”的主要来源。

- **基础增强**（必用）：
    
    旋转、平移、缩放、翻转、裁剪、亮度/对比度/HSV 扰动。
    
- **强增强**（小数据时很有用）：
    
    - **CutOut / Random Erasing**：随机盖掉一块区域，逼模型不依赖局部特征。
        
    - **MixUp / CutMix**：两张图按比例混合，标签也混合，增加“中间态”样本。
        
    - 对数字/字符：弹性形变、模拟笔迹粗细变化、运动模糊。
        
    

> 小数据下，增强做得越“离谱”但合理，往往收益越大。

---

### 2. 合成数据：自己造“假”样本

如果真实数据太少，可以**程序化生成**。

- 对数字识别：
    
    - 用不同字体、字号、颜色，在随机背景上贴数字；
        
    - 加各种噪声、模糊、透视变形，模拟真实拍摄。
        
    
- 对分类：
    
    - 用 GAN 或简单规则生成新图（如猫的不同姿态、不同角度）。
        
    

关键点：

- 合成数据要**尽量接近真实分布**；
    
- 可以先用合成数据预训练，再用真实数据微调。
    

---

### 3. 半监督 / 自监督：用“无标签”数据

如果你有**少量有标签 + 大量无标签**图，可以：

- 先用有标签数据训练一个简单模型；
    
- 用这个模型给无标签数据打伪标签，选置信度高的加入训练；
    
- 或者做**自监督预训练**：
    
    - 对无标签图做旋转预测、拼图、颜色变换预测等“代理任务”，学通用特征，再在小有标签集上微调。
        
    

---

### 二、再改模型：让模型“更会举一反三”

### 1. 用预训练模型（迁移学习）

这是小数据下**收益最大**的一招。

- 选一个在大数据集上预训练好的 Backbone（如 ImageNet 的 ResNet、EfficientNet）。
    
- 做法：
    
    1. 冻结前几层，只训练最后几层 + 自己的任务头；
        
    2. 等验证集效果稳定，再解冻部分层一起微调。
        
    

好处：

- 底层学到的是通用边缘、纹理，上层只需学你任务特有的模式，样本需求大大降低。
    

---

### 2. 模型要“小一点”

小数据配大模型 = 更容易背样本、过拟合。

- 能用 MobileNet/ShuffleNet 就不用超大 ResNet；
    
- 层数、通道数可以先砍一刀，保证训练稳定后再慢慢加。
    

---

### 3. 多任务学习（如果有相关任务）

除了主任务，还可以加一个**辅助任务**：

- 例如：数字识别 + 判断数字是否在图片中心；
    
- 或者：猫狗分类 + 背景纹理分类。
    

共享 Backbone，让模型从不同角度利用有限样本，学到更通用的特征。

---

### 4. 度量学习 / Few-shot 专用结构

如果真的是**极少量**（每个类 1～5 张）：

- 用 **Siamese Network、Prototypical Net、Relation Net**​ 这类结构；
    
- 核心思想：不再死记每张图，而是学一个“特征空间”，同类样本靠近、异类远离；
    
- 推理时靠“相似度匹配”，而不是单纯的分类层 softmax。
    

---

### 三、再调训练：用技巧稳住小样本训练

### 1. 正则化要足

- **Dropout**：在全连接层或注意力层加。
    
- **Weight Decay**：L2 正则，限制权重幅度。
    
- **Label Smoothing**：把 one-hot 标签稍微“软化”，防止过度自信。
    

---

### 2. 训练策略

- **冻结 + 解冻**：先冻 Backbone，只训任务头，稳定后再微调整体。
    
- **学习率要小**：小数据下大 lr 很容易震荡甚至发散。
    
- **早停**：监控验证集，一旦开始明显变差就停，防止过拟合。
    

---

### 3. 集成与测试时增强

- **Test-Time Augmentation (TTA)**：对测试图做多种增强，预测多次，再投票/平均。
    
- **模型集成**：训练几个不同初始化/不同结构的模型，结果平均，能显著提升小数据下的稳定性。
    

---

### 四、一个简单可执行的“小数据配方”

假设：每个数字只有 3 张训练图（你的题四那种）：

1. **数据侧**​
    
    - 用程序生成 500～1000 张合成数字图（不同字体、背景、噪声）；
        
    - 再做 MixUp/CutMix + Random Erasing。
        
    
2. **模型侧**​
    
    - Backbone：MobileNetV2（ImageNet 预训练）；
        
    - 冻结前 80% 层，只训练后半部分 + 一个新的 FC(10) 头；
        
    - 训练几个 epoch 稳定后，解冻全部层，用小 lr 微调。
        
    
3. **训练侧**​
    
    - 损失：交叉熵 + Label Smoothing；
        
    - 优化器：AdamW，weight decay 1e-4；
        
    - 早停：验证集准确率连续 5 epoch 不提升就停。
